// Exemple 1 : Logging personnalisé pour les fuites mémoire
    void MyLogCallback(const nk_char* message) {
        // Rediriger vers un fichier de log dédié
        FILE* f = fopen("memory_leaks.log", "a");
        if (f) {
            fprintf(f, "%s\n", message);
            fclose(f);
        }
        // Ou vers un système de logging externe
        ExternalLogger::Warn("Memory", message);
    }

    void InitLogging() {
        NK_MEMORY_SYSTEM.SetLogCallback(MyLogCallback);
    }

-Mon estimation :
*le module fournit un systeme de gestion de memoire unifie et securise pour le moteur. Il integre nativement du suivi (tracking) pour la detection de fuites, du profiling et une gestion centralisee du Garbage Collection, tout en restant simple a integrer via des macros.
